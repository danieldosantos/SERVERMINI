# LuckPerms command plan for D&D 5e structure
# Run these in-game or via console. Uses meta for PB, advantage flags, proficiencies, and ancestry/heritage bonuses.

# --- Core meta keys ---
# Set proficiency bonus per level (optional fallback; ValhallaMMO also maps automatically)
lp user <player> meta set pb 2

# Advantage/Disadvantage toggles (temporary, clear on rest)
lp user <player> meta set condition:advantage true
lp user <player> meta set condition:disadvantage true

# --- Races and subraces ---
# Example: Wood Elf with +2 DEX, +1 WIS and Darkvision
lp creategroup race_wood_elf
lp group race_wood_elf meta set bonus:DEX 2
lp group race_wood_elf meta set bonus:WIS 1
lp group race_wood_elf permission set essentials.effect.nightvision true
lp group race_wood_elf permission set prof.simple_weapons true

# Apply to player
lp user <player> parent add race_wood_elf
